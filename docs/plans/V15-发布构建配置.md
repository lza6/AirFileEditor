## build.gradle.kts 修改

在 `android {}` 块中，`buildTypes` 前添加 `signingConfigs`，并在 release 中引用：

```kotlin
android {
    // 在 defaultConfig 之后添加
    signingConfigs {
        create("release") {
            storeFile = file("${System.getenv("LOCALAPPDATA")}/Android/Sdk/tfgwj-release-key.jks")
            storePassword = "tfgwj2026"
            keyAlias = "tfgwj"
            keyPassword = "tfgwj2026"
        }
    }

    buildTypes {
        release {
            isMinifyEnabled = true   // 开启混淆
            isShrinkResources = true // 资源压缩
            signingConfig = signingConfigs.getByName("release")
            proguardFiles(
                getDefaultProguardFile("proguard-android-optimize.txt"),
                "proguard-rules.pro",
            )
        }
    }
}
```

## ProGuard 规则增强

在 `app/proguard-rules.pro` 末尾追加：

```proguard
# 保留 Shizuku 相关类
-keep class rikka.shizuku.** { *; }
-keep class dev.rikka.shizuku.** { *; }

# 保留 Zip4j 和 Commons-Compress（反射使用）
-keep class net.lingala.zip4j.** { *; }
-keep class org.apache.commons.compress.** { *; }
-keep class org.tukaani.xz.** { *; }

# 保留 JSON 序列化模型
-keep class com.example.tfgwj.data.** { *; }
-keep class com.example.tfgwj.domain.model.** { *; }
-keep class com.example.tfgwj.model.** { *; }

# 保留 AIDL 生成的接口
-keep interface com.example.tfgwj.IFileOperationService { *; }
-keep class com.example.tfgwj.IFileOperationService$** { *; }

# 保留 Service 和 Worker（WorkManager 反射）
-keep class com.example.tfgwj.shizuku.FileOperationService { *; }
-keep class com.example.tfgwj.worker.FileReplaceWorkerV2 { *; }

# 保留 ViewBinding/DataBinding 生成的类
-keep class com.example.tfgwj.databinding.** { *; }
-keep class com.example.tfgwj.viewbinding.** { *; }

# 保留 Compose 相关
-keep class androidx.compose.** { *; }

# 保留鸿蒙反射检测类
-keepclassmembers class * {
    *** *getOsBrand();
}
```
